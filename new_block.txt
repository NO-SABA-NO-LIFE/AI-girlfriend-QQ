def _provider_payload_overrides(base_url: str, model: str) -> dict[str, Any]:
    folded_url = str(base_url).casefold()
    folded_model = str(model).strip().casefold()
    if (
        "api.deepseek.com" in folded_url
        and folded_model in {"deepseek-v4-flash", "deepseek-v4-pro"}
    ):
        return {"thinking": {"type": "disabled"}}
    # higanai relays Gemini 3.x models; their default reasoning can consume
    # the whole completion budget and truncate the visible reply. The gateway
    # honors the OpenAI-style reasoning_effort parameter, so keep hidden
    # thinking short while retaining the persona quality.
    if "api.higanai.top" in folded_url:
        return {"reasoning_effort": "low"}
    return {}
